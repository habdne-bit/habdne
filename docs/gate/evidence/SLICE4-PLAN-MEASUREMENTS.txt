def outcome(fn):
    try:
        return f"OK: {fn()}"
    except psycopg.Error as e:

def section(title):

section("D. plan §3.3: two runs, same (request, property, policy, input_hash)")


def rr_case(label, statement_suffix, commit_first):
    h = f"probe-D-{uuid.uuid4().hex[:8]}"
    t1 = conn(psycopg.IsolationLevel.REPEATABLE_READ)
    t1.execute("SELECT 1")  # the snapshot is taken here
    t2 = conn()
    insert_match(t2, h)
    if commit_first:
        t2.commit()
        result = outcome(lambda: insert_match(t1, h, statement_suffix))
    else:
        box = {}

        def contend():
            box["r"] = outcome(lambda: insert_match(t1, h, statement_suffix))
        th = threading.Thread(target=contend)
        th.start()
        pid1 = t1.info.backend_pid
        waited = False
        for _ in range(100):
            blockers = admin.execute("SELECT pg_blocking_pids(%s)", (pid1,)).fetchone()[0]
            if blockers:
                waited = True
                break
            time.sleep(0.05)
        t2.commit()
        th.join()
        result = f"(waited on the uncommitted row: {waited}) {box['r']}"
    t1.rollback()
    t1.execute("SELECT 1")
    print(f"{label}: {result}")
    t1.close()
    t2.close()
    return h
