"""Slice 5 plan, revision 2: measure the facts its decisions rest on, before
any Slice 5 code exists.

def connect(autocommit=False) -> psycopg.Connection:
    c = psycopg.connect(CONNINFO, autocommit=autocommit)

SETUP = connect(autocommit=True)


def one(sql, **p):
    return SETUP.execute(sql, p).fetchone()[0]


def rows(sql, **p):
    cur = SETUP.execute(sql, p)
    names = [d.name for d in cur.description]
    return [dict(zip(names, r)) for r in cur.fetchall()]


def section(title):
    print(f"\n{'=' * 78}\n{title}\n{'=' * 78}")


def attempt(c, label, sql, params=None, keep=False):
    """Run one statement in a savepoint. Rolled back unless `keep`."""
    c.execute("SAVEPOINT probe")
    try:
        c.execute(sql, params or {})
        if keep:
            c.execute("RELEASE SAVEPOINT probe")
        else:
            c.execute("ROLLBACK TO SAVEPOINT probe")
        print(f"  {label:66} ACCEPTED")
        return "ACCEPTED"
    except psycopg.Error as e:
        c.execute("ROLLBACK TO SAVEPOINT probe")
        d = e.diag
        print(f"  {label:66} REFUSED {e.sqlstate}"
              + (f" [{d.constraint_name}]" if d.constraint_name else "")
              + f": {d.message_primary}")
        MESSAGES.append((label, e.sqlstate, d.message_primary, d.constraint_name,
                         d.context.splitlines()[0] if d.context else ""))
        return "REFUSED"

def run(req, props):
    r = client.post(f"/requests/{req}/matching/run",
                    headers={"Authorization": f"Bearer {ACC_OPERATOR}",
                             "Idempotency-Key": str(uuid.uuid4())},
                    json={"matching_policy_version": POLICY,
                          "property_ids": [str(p) for p in props]})
    assert r.status_code == 201, r.text
    return r.json()["matches"]


def as_reviewer(c):
    c.execute("SELECT set_config('app.account_id', %s, true)", (ACC_REVIEWER,))


REVIEW = """INSERT INTO match_reviews (match_id, decision, reason_code, reviewer_account_id)
            VALUES (%(m)s, %(d)s, %(r)s, %(a)s) RETURNING match_review_id"""
OPPORTUNITY = """
    INSERT INTO opportunities (request_id, property_id, approved_match_id, current_offer_id,
                               sharing_scope, why_real, created_by_account_id)
    SELECT request_id, property_id, match_id, evaluated_offer_id, 'SUMMARY_ONLY',
           '{}'::jsonb, %(a)s
      FROM match_candidates WHERE match_id = %(m)s"""


def review(match_id, decision, reason=None):
    with connect() as c:
        as_reviewer(c)
        rid = c.execute(REVIEW, {"m": match_id, "d": decision, "r": reason,
                                 "a": ACC_REVIEWER}).fetchone()[0]
        c.commit()
    return rid


def open_opportunity(match_id):
    with connect() as c:
        as_reviewer(c)
        c.execute(OPPORTUNITY + " RETURNING opportunity_id",
                  {"m": match_id, "a": ACC_REVIEWER})
        oid = c.execute("SELECT opportunity_id FROM opportunities "
                        "WHERE approved_match_id = %s", (match_id,)).fetchone()[0]
        c.commit()
    return oid

    git = lambda *a: subprocess.run(["git", "-C", ROOT, *a], capture_output=True,
                                    text=True).stdout.strip()
    print("Database :", DB, "|", one("SELECT version()").split(",")[0])
    print("Commit   :", git("rev-parse", "HEAD"))
    print("Migrations:", one("SELECT version_num FROM alembic_version"))

    # ===================================================================== H
    section("H  Review point 1 (G5-2, G5-11): two offers of one property, one run")

    ms = rows("""SELECT match_id, evaluated_offer_id, matching_policy_id, evaluated_at,
                        created_at, input_hash, eligibility::text AS eligibility

          len({(req, p, m["matching_policy_id"]) for m in ms}) == 1)

          "| created_at equal:", ms[0]["created_at"] == ms[1]["created_at"])
    SETUP.execute("UPDATE property_offers SET asking_price_dzd = 21000000 "
                  "WHERE offer_id = %s", (o2,))
    time.sleep(0.01)
    run(req, [p])
    ms = rows("""SELECT match_id, evaluated_offer_id, evaluated_at
                   FROM match_candidates WHERE request_id = %(r)s
                  ORDER BY evaluated_at, match_id""", r=req)
    print(f"  after O2's price changes and a second run: {len(ms)} matches")
    latest_pair = rows("""SELECT DISTINCT ON (request_id, property_id, matching_policy_id)
                                 match_id, evaluated_offer_id
                            FROM match_candidates WHERE request_id = %(r)s
                           ORDER BY request_id, property_id, matching_policy_id,
                                    evaluated_at DESC, match_id DESC""", r=req)

                                  evaluated_offer_id
                             FROM match_candidates WHERE request_id = %(r)s

    name = lambda o: "O1" if o == o1 else "O2"
    print("  'latest per (request, property, policy)' keeps:",

    print("  'latest per (request, property, policy, offer)' keeps:",
          sorted(name(m["evaluated_offer_id"]) for m in latest_offer))
    print("  run ONCE on two offers, latest-per-pair ordering ties on evaluated_at and")
    print("  falls to match_id, a random uuid: the survivor is arbitrary.")

    # ===================================================================== A
    section("A  §6.4.1: the latest-review ordering of trg_opportunity_gate")
    req = request()
    p = prop()
    offer(p)
    [m] = run(req, [p])
    mid = m["match_id"]
    print(f"  match {mid[:8]}: eligibility={m['eligibility']}")
    review(mid, "APPROVED")
    time.sleep(0.01)
    review(mid, "NEED_MORE_INFORMATION", "DOCUMENT_NOT_KNOWN")
    with connect() as c:
        as_reviewer(c)
        attempt(c, "A1 APPROVED, then NMI (later transaction): insert opportunity",
                OPPORTUNITY, {"m": mid, "a": ACC_REVIEWER})
        c.rollback()
    time.sleep(0.01)
    review(mid, "APPROVED")
    with connect() as c:
        as_reviewer(c)
        attempt(c, "A2 ... then APPROVED again: insert opportunity",
                OPPORTUNITY, {"m": mid, "a": ACC_REVIEWER})
        c.rollback()
    print("  reviews on the match:", [r["decision"] for r in rows(
        "SELECT decision::text AS decision FROM match_reviews WHERE match_id = %(m)s "
        "ORDER BY reviewed_at, match_review_id", m=mid)], "(append-only, all kept)")

    print("\n  A3 APPROVED then NMI inserted in ONE transaction (same now()), 20 trials,")
    print("     each rolled back; the trigger's 'latest' is reviewed_at DESC,")
    print("     match_review_id DESC:")
    req = request()
    p = prop()
    offer(p)
    [m] = run(req, [p])
    mid = m["match_id"]
    outcome = {"opportunity ACCEPTED (APPROVED read as latest)": 0,
               "opportunity REFUSED (NMI read as latest)": 0}
    same_time = 0
    for _ in range(20):
        with connect() as c:
            as_reviewer(c)
            a = c.execute(REVIEW, {"m": mid, "d": "APPROVED", "r": None,
                                   "a": ACC_REVIEWER}).fetchone()[0]
            n = c.execute(REVIEW, {"m": mid, "d": "NEED_MORE_INFORMATION",
                                   "r": "DOCUMENT_NOT_KNOWN", "a": ACC_REVIEWER}).fetchone()[0]
            t = c.execute("SELECT count(DISTINCT reviewed_at) FROM match_reviews "
                          "WHERE match_review_id IN (%s, %s)", (a, n)).fetchone()[0]
            same_time += t == 1
            c.execute("SAVEPOINT s")
            try:
                c.execute(OPPORTUNITY, {"m": mid, "a": ACC_REVIEWER})
                outcome["opportunity ACCEPTED (APPROVED read as latest)"] += 1
            except psycopg.Error:
                c.execute("ROLLBACK TO SAVEPOINT s")
                outcome["opportunity REFUSED (NMI read as latest)"] += 1
            c.rollback()
    print(f"     both reviews had the same reviewed_at in {same_time}/20 trials")
    for k, v in outcome.items():
        print(f"     {k}: {v}/20")
    print("     The NMI was always inserted SECOND. Which one the trigger reads as")
    print("     latest is decided by the random uuid, not by the insertion order.")

    print("\n  A5 two review transactions on one match: B STARTS first, A inserts NMI and")
    print("     commits, THEN B inserts APPROVED and commits (B is the last decision):")
    req = request()
    p = prop()
    offer(p)
    [m] = run(req, [p])
    mid = m["match_id"]
    cb, ca = connect(), connect()
    as_reviewer(cb)
    cb.execute("SELECT now()")
    time.sleep(0.05)
    as_reviewer(ca)
    ca.execute(REVIEW, {"m": mid, "d": "NEED_MORE_INFORMATION", "r": "DOCUMENT_NOT_KNOWN",
                        "a": ACC_REVIEWER})
    ca.commit()
    cb.execute(REVIEW, {"m": mid, "d": "APPROVED", "r": None, "a": ACC_REVIEWER})
    cb.commit()
    ca.close()
    cb.close()
    order = rows("""SELECT decision::text AS decision, reviewed_at FROM match_reviews
                     WHERE match_id = %(m)s ORDER BY reviewed_at DESC, match_review_id DESC""",
                 m=mid)
    print(f"     committed order: NMI, then APPROVED; the trigger's order (latest first): "
          f"{[r['decision'] for r in order]}")
    with connect() as c:
        as_reviewer(c)
        attempt(c, "A5 insert opportunity after APPROVED was committed last", OPPORTUNITY,
                {"m": mid, "a": ACC_REVIEWER})
        c.rollback()

    print("\n  A6 a row lock on the immutable match row:")
    c1, c2 = connect(), connect()
    attempt(c1, "A6 SELECT ... FOR UPDATE on match_candidates",
            "SELECT match_id FROM match_candidates WHERE match_id = %(m)s FOR UPDATE",
            {"m": mid}, keep=True)
    pid1 = c1.execute("SELECT pg_backend_pid()").fetchone()[0]
    pid2 = c2.execute("SELECT pg_backend_pid()").fetchone()[0]
    got = {}

    def lock2():
        c2.execute("SELECT match_id FROM match_candidates WHERE match_id = %s FOR UPDATE",
                   (mid,))
        got["t"] = c2.execute("SELECT clock_timestamp() > now()").fetchone()[0]

    t = threading.Thread(target=lock2)
    t.start()
    waited = False
    for _ in range(100):
        time.sleep(0.02)
        if pid1 in one("SELECT pg_blocking_pids(%(p)s)", p=pid2):
            waited = True
            break
    c1.rollback()
    t.join(10)
    print(f"     a second FOR UPDATE waits on the first: {waited}; once it holds the lock,"
          f" clock_timestamp() > now(): {got.get('t')}")
    c2.rollback()
    c1.close()
    c2.close()

    req = request()
    p = prop(document=None)
    offer(p)
    [m] = run(req, [p])
    with connect() as c:
        as_reviewer(c)
        attempt(c, f"A4 APPROVED on a {m['eligibility']} match (trg_match_review_gate)",
                REVIEW, {"m": m["match_id"], "d": "APPROVED", "r": None, "a": ACC_REVIEWER})
        c.rollback()

    # ===================================================================== B
    section("B  §6.4.2: what the schema lets an UPDATE do to an opportunity")
    req = request()
    p = prop()
    o = offer(p)
    o_other = offer(p)
    [ma, mb] = sorted(run(req, [p]), key=lambda m: m["evaluated_offer_id"] != str(o))
    review(ma["match_id"], "APPROVED")
    opp = open_opportunity(ma["match_id"])
    other_req = request()
    print(f"  opportunity {str(opp)[:8]} on match {ma['match_id'][:8]} (offer O), "
          "status NEW, open")
    cases = [
        ("B1 request_id -> another request", "request_id = %(x)s", other_req),
        ("B2 property_id -> another property", "property_id = %(x)s", prop()),
        ("B3 approved_match_id -> the pair's other (unapproved) match",
         "approved_match_id = %(x)s", mb["match_id"]),
        ("B4 commercial_context_snapshot rewritten",
         "commercial_context_snapshot = '{\"rewritten\": true}'::jsonb", None),
        ("B5 permission_snapshot rewritten",
         "permission_snapshot = '{\"rewritten\": true}'::jsonb", None),
        ("B6 why_real rewritten", "why_real = '{\"rewritten\": true}'::jsonb", None),
        ("B7 sharing_scope SUMMARY_ONLY -> CONTACT_AFTER_CONFIRMATION",
         "sharing_scope = 'CONTACT_AFTER_CONFIRMATION'", None),
        ("B8 created_by_account_id -> the operator", "created_by_account_id = %(x)s",
         ACC_OPERATOR),
        ("B9 created_at moved a year back", "created_at = created_at - interval '1 year'",
         None),
        ("B10 current_offer_id -> the property's other offer", "current_offer_id = %(x)s",
         o_other),
        ("B11 status NEW -> ENGAGED (skips SHARED)", "status = 'ENGAGED'", None),
        ("B12 status NEW -> CLOSED with no closed_at", "status = 'CLOSED'", None),
        ("B13 closed_at set while NEW", "closed_at = now()", None),
    ]
    with connect() as c:
        as_reviewer(c)
        for label, assignment, x in cases:
            attempt(c, label, f"UPDATE opportunities SET {assignment} "
                              "WHERE opportunity_id = %(o)s", {"o": opp, "x": x})
        c.execute("SAVEPOINT closed")
        c.execute("UPDATE opportunities SET status = 'CLOSED', closed_at = now(), "
                  "close_reason_code = 'BUYER_REJECTED' WHERE opportunity_id = %s", (opp,))
        attempt(c, "B14 CLOSED -> NEW (reopened)", "UPDATE opportunities SET status = 'NEW', "
                "closed_at = NULL WHERE opportunity_id = %(o)s", {"o": opp})
        attempt(c, "B15 CLOSED -> SHARED", "UPDATE opportunities SET status = 'SHARED' "
                "WHERE opportunity_id = %(o)s", {"o": opp})
        c.execute("ROLLBACK TO SAVEPOINT closed")
        c.execute("UPDATE opportunities SET status = 'SHARED', shared_at = now() "
                  "WHERE opportunity_id = %s", (opp,))
        attempt(c, "B16 SHARED -> NEW (backwards)", "UPDATE opportunities SET status = 'NEW' "
                "WHERE opportunity_id = %(o)s", {"o": opp})

                "SET shared_at = NULL WHERE opportunity_id = %(o)s", {"o": opp})
        attempt(c, "B18 DELETE the opportunity", "DELETE FROM opportunities "
                "WHERE opportunity_id = %(o)s", {"o": opp})
        attempt(c, "B19 DELETE a review (prevent_match_review_update)",
                "DELETE FROM match_reviews WHERE match_id = %(m)s", {"m": ma["match_id"]})
        c.rollback()
    print("  (each case alone, from the same NEW row, rolled back)")

    # ===================================================================== C
    section("C  §6.4.3: two concurrent inserts for one pair")
    for isolation in ("READ COMMITTED", "REPEATABLE READ"):
        req = request()
        p = prop()

        review(m2["match_id"], "APPROVED")
        c1, c2 = connect(), connect()
        for c in (c1, c2):
            c.execute(f"SET TRANSACTION ISOLATION LEVEL {isolation}")
            as_reviewer(c)
        pid1 = c1.execute("SELECT pg_backend_pid()").fetchone()[0]
        pid2 = c2.execute("SELECT pg_backend_pid()").fetchone()[0]
        c2.execute("SELECT 1")  # take the snapshot before c1 inserts (RR)
        c1.execute(OPPORTUNITY, {"m": m1["match_id"], "a": ACC_REVIEWER})
        result = {}

        def second():
            try:
                c2.execute(OPPORTUNITY, {"m": m2["match_id"], "a": ACC_REVIEWER})
                result["r"] = "ACCEPTED"
            except psycopg.Error as e:
                result["r"] = f"REFUSED {e.sqlstate} [{e.diag.constraint_name}]: " \
                              f"{e.diag.message_primary}"
                MESSAGES.append((f"C {isolation} second insert", e.sqlstate,
                                 e.diag.message_primary, e.diag.constraint_name, ""))

        t = threading.Thread(target=second)
        t.start()
        blocked = False
        for _ in range(100):
            time.sleep(0.02)
            bp = one("SELECT pg_blocking_pids(%(p)s)", p=pid2)
            if pid1 in bp:
                blocked = True
                break
        c1.commit()
        t.join(10)
        c2.rollback()
        print(f"  {isolation}: second insert waited on the first "
              f"(pg_blocking_pids({pid2}) contains {pid1}): {blocked}")
        print(f"  {isolation}: after the first commits, the second: {result.get('r')}")
        c1.close()
        c2.close()

    # ===================================================================== F
    section("F  §6.4.6: a closed opportunity frees the pair, not its match")

                {"m": m2["match_id"], "a": ACC_REVIEWER}, keep=True)

                  "close_reason_code = 'OTHER' WHERE approved_match_id = %s",
                  (m2["match_id"],))

                {"m": m1["match_id"], "a": ACC_REVIEWER})

          "| rolsuper:", one("SELECT rolsuper FROM pg_roles WHERE rolname = current_user"),
          "| rolbypassrls:", one("SELECT rolbypassrls FROM pg_roles "
                                 "WHERE rolname = current_user"))
    for t in ("match_reviews", "opportunities", "tasks", "interactions",
              "task_completion_events", "opportunity_responses"):
        privs = {pv: one("SELECT has_table_privilege(%(t)s, %(p)s)", t=f"turab.{t}", p=pv)
                 for pv in ("INSERT", "UPDATE", "DELETE", "TRUNCATE")}
        print(f"  {t:24} owner={one('SELECT tableowner FROM pg_tables WHERE schemaname = %(s)s AND tablename = %(t)s', s='turab', t=t)} "
              + " ".join(f"{k}={v}" for k, v in privs.items()))
    print("  A superuser bypasses every GRANT. Only triggers stand between the role")
    print("  and these tables (and a superuser can disable triggers).")

    # ===================================================================== E

             WHERE tgrelid = %(t)s::regclass AND NOT tgisinternal ORDER BY tgname""",
                                         t=f"turab.{t}")]
        print(f"  {t:24} triggers: {trg}")
    for t in ("match_reviews", "opportunities"):
        a = rows("""SELECT action, count(*) AS n, count(actor_account_id) AS with_actor
                      FROM audit_log WHERE entity_table = %(t)s
                     GROUP BY action ORDER BY action""", t=t)
        print(f"  audit_log {t}: {a}")
    req = request()
    p = prop()
    offer(p)
    [m] = run(req, [p])
    with connect() as c:
        attempt(c, "E1 review inserted with app.account_id UNSET", REVIEW,

        SELECT column_name FROM information_schema.columns

    print("  a column marking a claim private or public:",
          [c for c in cols if any(w in c for w in ("private", "visib", "public", "share"))]
          or "none")
    import yaml
    S = yaml.safe_load(open(os.path.join(ROOT, "docs/handoff/05_API/openapi_v0.2.3.yaml")))[

    pps = set(S["PublicPropertySummary"]["properties"])

             "LAND_AREA_MIN": "land_area_m2", "BUILT_AREA_MIN": "built_area_m2",
             "ROOMS_MIN": "attribute ROOMS", "BEDROOMS_MIN": "attribute BEDROOMS",
             "DOCUMENT_TYPE": "attribute DOCUMENT_TYPE", "RIGHT_TYPE": "attribute RIGHT_TYPE",

    section("G  §6.4.7: every refusal's SQLSTATE and message, as the database gives them")
    seen = set()
    for label, state, msg, constraint, ctx in MESSAGES:
        key = (state, msg, constraint)
