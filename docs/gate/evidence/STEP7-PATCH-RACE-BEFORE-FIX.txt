"""TEMPORARY measurement (review of 1935dc1): does a PATCH committed in
generation's window leave a candidate whose blocking facts no longer hold?"""
import threading, time, uuid
import pytest
from sqlalchemy import text
from sqlalchemy.orm import Session
from tests.test_slice3_identity import client, sink, _location, _property, _h, _all  # noqa: F401
from turab.db.session import audited_transaction
from turab.services import identity


def _generate_paused_at(engine, ids, monkeypatch, name, focus):
    in_window, go_on, out = threading.Event(), threading.Event(), {}
    original = getattr(identity, name)
    def paused(*a, **k):
        if not in_window.is_set():
            in_window.set(); go_on.wait(30)
        return original(*a, **k)
    monkeypatch.setattr(identity, name, paused)
    def run():
        try:
            with Session(bind=engine, future=True) as s, audited_transaction(s, ids.ACC_OPERATOR):
                out["pid"] = s.execute(text("SELECT pg_backend_pid()")).scalar_one()
                rows = identity.generate(s, property_id=focus, algorithm_version=None)
            out["rows"] = [(str(r["property_a_id"]), str(r["property_b_id"])) for r in rows]
        except Exception as exc:
            out["rows"] = f"raised {type(exc).__name__}: {exc}"
    t = threading.Thread(target=run); t.start()
    return t, in_window, go_on, out


@pytest.mark.parametrize("window", ["a_before_lock", "b_after_lock"])
@pytest.mark.parametrize("change", ["property_type", "canonical_location_id"])
def test_measure(client, ids, engine, monkeypatch, window, change):
    loc = _location(engine)
    b, c = _property(client, ids, loc), _property(client, ids, loc)
    name = "_lock_and_recheck" if window == "a_before_lock" else "_candidate_for_pair"
    t, in_window, go_on, out = _generate_paused_at(engine, ids, monkeypatch, name, uuid.UUID(c))
    assert in_window.wait(30)
    body = {"property_type": "APARTMENT"} if change == "property_type" \
        else {"canonical_location_id": str(_location(engine))}
    patch = {}
    def do_patch():
        r = client.patch(f"/properties/{b}", headers={**_h(ids.ACC_OPERATOR), "If-Match-Version": "1"},
                         json=body)
        patch["status"] = r.status_code; patch["at"] = time.monotonic()
    pt = threading.Thread(target=do_patch); pt.start()
    waited = False
    deadline = time.monotonic() + 5
    while time.monotonic() < deadline and pt.is_alive():
        with Session(bind=engine, future=True) as s:
            blocked = s.execute(text("SELECT count(*) FROM pg_stat_activity "
                                     "WHERE :g = ANY(pg_blocking_pids(pid))"), {"g": out["pid"]}).scalar_one()
        if blocked:
            waited = True; break
        time.sleep(0.01)
    patch_done_before_release = not pt.is_alive()
    go_on.set(); t.join(30); pt.join(30)
    rows = _all(engine, """SELECT p.property_type::text AS b_type, p.canonical_location_id = q.canonical_location_id AS same_loc,
                                  p.property_type = q.property_type AS same_type, c.signals->'blocking' AS claimed
                             FROM turab.property_identity_candidates c
                             JOIN turab.properties p ON p.property_id = :b
                             JOIN turab.properties q ON q.property_id = :c
                            WHERE LEAST(c.property_a_id, c.property_b_id) = LEAST(:b, :c)
                              AND GREATEST(c.property_a_id, c.property_b_id) = GREATEST(:b, :c)""",
                b=b, c=c)
    print(f"\nMEASURE window={window} change={change} patch_waited_on_generation={waited} "
          f"patch_committed_inside_window={patch_done_before_release} patch_status={patch.get('status')} "
          f"generated={out['rows']} candidate_now={rows}")
