Usage: PYTHONPATH=<tree>/src python measure.py"""
import decimal
import uuid
from decimal import Decimal

from turab.matching import canonical
from turab.matching.snapshots import exact_json

def h(request_snapshot):
    return canonical.input_hash(
        matching_policy_id=uuid.UUID(int=1), matching_policy_version="0.2.0",
        rule_registry_digest="d" * 64, evaluated_offer_id=uuid.UUID(int=2),
        request_snapshot=request_snapshot, property_snapshot={},
        commercial_context_snapshot={}, permission_snapshot={}, freshness_snapshot={})

def show(label, a, b):
    ba, bb = canonical.canonical_bytes(a), canonical.canonical_bytes(b)
    print(f"{label}\n  bytes A: {ba!r}\n  bytes B: {bb!r}\n  same bytes: {ba == bb}; "
          f"same input_hash: {h(a) == h(b)}")

print(f"format tag: {canonical.INPUT_FORMAT}; decimal context precision: {decimal.getcontext().prec}")
show("1. JSON number 1 (read by exact_json) vs JSON string \"1\"",
     {"value": exact_json("1")}, {"value": exact_json('"1"')})
a, b = Decimal("1.00000000000000000000000000001"), Decimal("1.00000000000000000000000000002")
show("2. two Decimals differing in the 30th significant digit, default context",
     {"value": a}, {"value": b})
with decimal.localcontext() as ctx:
    ctx.prec = 40
    show("2b. the same two Decimals, context precision 40", {"value": a}, {"value": b})
print("3. a criterion value {\"evaluated_at\": \"2012-03-04\"} inside the request snapshot:")
try:
    print("  hash:", h({"criteria": [{"value": {"evaluated_at": "2012-03-04"}}]}))
except canonical.CanonicalError as e:
    print("  REFUSED:", e)
