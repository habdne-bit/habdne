def one(sql, **p):
    return SETUP.execute(sql, p).fetchone()[0]


def rows(sql, **p):
    cur = SETUP.execute(sql, p)
    names = [d.name for d in cur.description]
    return [dict(zip(names, r)) for r in cur.fetchall()]


def section(title):

def attempt(c, label, sql, params=None):
    """One statement in a savepoint, always rolled back."""
    c.execute("SAVEPOINT probe")
    try:
        cur = c.execute(sql, params or {})
        n = cur.rowcount
        c.execute("ROLLBACK TO SAVEPOINT probe")
        print(f"  {label:64} ACCEPTED (rows={n})")
        return "ACCEPTED"
    except psycopg.Error as e:
        c.execute("ROLLBACK TO SAVEPOINT probe")
        d = e.diag
        print(f"  {label:64} REFUSED {e.sqlstate}"
              + (f" [{d.constraint_name}]" if d.constraint_name else "")
              + f": {d.message_primary}")
        return "REFUSED"

def chain(location_id):
    """The location and its ancestors, nearest first, from the CURRENT table."""
    return [r["code"] for r in rows("""
        WITH RECURSIVE up(location_id, parent_id, code, location_type, depth) AS (
            SELECT location_id, parent_id, code, location_type, 0 FROM locations
             WHERE location_id = %(l)s
            UNION ALL
            SELECT l.location_id, l.parent_id, l.code, l.location_type, up.depth + 1
              FROM locations l JOIN up ON l.location_id = up.parent_id)
        SELECT code FROM up ORDER BY depth""", l=location_id)]

            INSERT INTO property_attributes (property_id, attribute_definition_id, value)
            SELECT %(p)s, attribute_definition_id, %(v)s
              FROM attribute_definitions WHERE code = 'DOCUMENT_TYPE'""",
                      {"p": p, "v": json.dumps(document)})
    return p


def offer(p, *, price=20_000_000, negotiable="NO", terms="1 day"):
    o = one(f"""
        INSERT INTO property_offers (property_id, party_id, transaction_type, status,
                                     asking_price_dzd, price_negotiable, permission_scope,
                                     commercial_terms_last_confirmed_at)
        VALUES (%(p)s, %(party)s, 'SALE', 'ACTIVE', %(price)s, %(n)s, 'SUMMARY_ONLY',
                now() - interval '{terms}')
        RETURNING offer_id""", p=p, party=BRAHIM, price=price, n=negotiable)
    grant = one("""
        INSERT INTO consent_grants (party_id, scope, channel, consent_version, granted_at)
        VALUES (%(party)s, 'PRIVATE_MATCHING_ONLY', 'PHONE_CONFIRMED', 'v1',
                now() - interval '2 days')
        RETURNING consent_id""", party=BRAHIM)
    SETUP.execute("""
        INSERT INTO resource_consent_bindings (consent_id, purpose, offer_id, bound_at)
        VALUES (%(c)s, 'PRIVATE_MATCHING_ONLY', %(o)s, now() - interval '1 day')""",
                  {"c": grant, "o": o})
    return o


def run(req, props):
    r = client.post(f"/requests/{req}/matching/run",
                    headers={"Authorization": f"Bearer {ACC_OPERATOR}",
                             "Idempotency-Key": str(uuid.uuid4())},
                    json={"matching_policy_version": POLICY,
                          "property_ids": [str(p) for p in props]})
    assert r.status_code == 201, r.text
    return r.json()


def criterion_row(match_id, code):
    return rows("""SELECT compatibility::text AS compatibility, request_value, property_value,
                          reason_code, rule_id, rule_version
                     FROM match_criterion_results
                    WHERE match_id = %(m)s AND criterion_code = %(c)s""",
                m=match_id, c=code)[0]

        print(f"     property snapshot location_ancestry = "
              f"{[code_of(x) for x in snap['location_ancestry']]}")
        pchain = chain(loc(here))
        for w in wanted:
            wchain = chain(loc(w))
            common = next((c for c in pchain if c in wchain), None)
            print(f"     requested {w:18} chain={wchain}")
            print(f"        nearest common ancestor = {common} ({typ(common) if common else '-'}),"
                  f" requested widened by {wchain.index(common) if common else '-'} level(s),"
                  f" property {pchain.index(common) if common else '-'} level(s) below it")
        # The ancestors of a requested location that are NOT on the property's
        # own chain: if the run stored a requested chain, they would appear in
        # the match row, its criterion rows or the run's diagnostic row.
        only_requested = sorted({a for w in wanted for a in chain(loc(w))[1:]} - set(pchain))
        stored_text = one("SELECT to_jsonb(m)::text FROM match_candidates m "
                          "WHERE match_id = %(m)s", m=m) + "".join(
            r["t"] for r in rows("SELECT to_jsonb(c)::text AS t FROM match_criterion_results c "
                                 "WHERE match_id = %(m)s", m=m)) + "".join(

        print(f"     ancestors of a requested location off the property's chain: "
              f"{only_requested or 'none'}; found in the match, criterion or diagnostic"
              f" rows: {found or ('none' if only_requested else 'n/a')}")

    print("\n  L5 are the chains fixed? an UPDATE of a requested location's parent,"

    with connect() as c:

    trig = [r["tgname"] for r in rows("""SELECT tgname FROM pg_trigger
                                          WHERE tgrelid = 'turab.locations'::regclass
                                            AND NOT tgisinternal""")]

    # ===================================================================== D
    section("D  Review point 2 (G6-11): which matches a diagnostic row evaluated")
    req = request([("DOCUMENT_TYPE", "EQ", "LAND_BOOK", "REQUIRED")])

    second = run(req, [p1, p2])
    m2 = {m["property_id"]: m["match_id"] for m in second["matches"]}

          f"(new: {m1[str(p2)] != m2[str(p2)]})")
    diags = rows("""SELECT diagnostic_run_id, run_at, input_hash, blocker_summary
                      FROM match_diagnostic_runs WHERE request_id = %(r)s ORDER BY run_at""",

    cols = [r["column_name"] for r in rows("""
        SELECT column_name FROM information_schema.columns

    print(f"  match_diagnostic_runs columns: {cols}")
    print("  a column naming a match: "
          f"{[c for c in cols if 'match' in c and 'policy' not in c] or 'none'}")
    d2 = diags[1]
    same_instant = rows("""SELECT property_id, match_id FROM match_candidates
                            WHERE request_id = %(r)s AND evaluated_at = %(t)s""",
                        r=req, t=d2["run_at"])
    print(f"  matches with evaluated_at = run 2's run_at: "
          f"{[(('P1' if str(x['property_id']) == str(p1) else 'P2'), str(x['match_id'])[:8]) for x in same_instant]}")
    print("  -> the reused P1 match carries run 1's evaluated_at; 'the matches of run 2'"
          " cannot be read from evaluated_at")
    all_ms = rows("SELECT match_id FROM match_candidates WHERE request_id = %(r)s", r=req)
    print(f"  matches stored for the request: {len(all_ms)} (run 2 evaluated 2 of them)")
    print(f"  run 2's input_hash: {d2['input_hash'][:16]}... (sha256 of a document listing"
          " [property, offer, match input_hash] per candidate; the document is not stored)")
    print(f"  exclusions stored in blocker_summary.excluded: "
          f"{'excluded' in d2['blocker_summary']} | excluded_counts: "
          f"{d2['blocker_summary'].get('excluded_counts')}")

    print("\n  D4 which row is 'latest': two diagnostic inserts in overlapping"
          " transactions (plain SQL, kept in the scratch database)")
    pol = one("SELECT matching_policy_id FROM matching_policies WHERE version = %(v)s",
              v=POLICY)
    ins = """INSERT INTO match_diagnostic_runs (request_id, request_version,
                 matching_policy_id, matching_policy_version, request_snapshot, input_hash,
                 run_at)
             VALUES (%(r)s, 1, %(p)s, %(v)s, '{}'::jsonb, %(h)s, transaction_timestamp())
             RETURNING diagnostic_run_id"""
    req4 = request([])
    b = connect()
    b.execute("SELECT 1")  # B's transaction starts first
    time.sleep(0.02)
    a = connect()
    a_id = a.execute(ins, {"r": req4, "p": pol, "v": POLICY, "h": "A"}).fetchone()[0]
    a.commit()

    b.commit()

                     ORDER BY run_at DESC, diagnostic_run_id DESC LIMIT 1""", r=req4)
    print(f"  committed order: A then B; the read's 'latest' (run_at DESC): {latest}")
    print("  -> run_at is the transaction START: the run committed last is not"

    # ===================================================================== T
    section("T  Review point 3 (G6-4 (iii)): the subject of task work, and its key")
    req = request([("DOCUMENT_TYPE", "EQ", "LAND_BOOK", "REQUIRED")], budget=20_000_000)
    p = prop()
    oa = offer(p, price=22_000_000, negotiable="YES")
    ob = offer(p, price=23_000_000, negotiable="YES", terms="40 days")
    body = run(req, [p])
    print("  one property, two offers, each above the maximum and negotiable;"
          " offer B's terms 40 days old")
    subjects = []
    for m in sorted(body["matches"], key=lambda x: x["evaluated_offer_id"] == str(ob)):
        name = "A" if m["evaluated_offer_id"] == str(oa) else "B"
        na = m["next_action"]
        reasons = rows("SELECT explanation -> 'reasons' AS r FROM match_candidates "
                       "WHERE match_id = %(m)s", m=m["match_id"])[0]["r"]
        print(f"  match on offer {name}: eligibility={m['eligibility']}")
        print(f"     next_action: type={na['type']} priority={na['priority']} "
              f"entity={na['subject']['entity']} id=offer {name} "
              f"criterion={na['subject'].get('criterion')} basis={na['subject']['basis']}")
        for r in reasons:
            print(f"     reason: gate={r['gate']} subject={r.get('subject')} "
                  f"criterion={r.get('criterion')} code={r['reason_code']} basis={r['basis']}")
        subjects.append((name, m, na))
    narrow = {(str(req), str(p), na["type"], na["subject"].get("criterion"),
               na["subject"].get("ordinal")) for _, _, na in subjects}
    wide = {(na["subject"]["entity"], na["subject"]["id"], na["subject"]["gate"],
             na["subject"].get("criterion"), na["subject"].get("ordinal"),
             na["subject"]["basis"]) for _, _, na in subjects}
    print(f"  key (request, property, type, criterion, ordinal): {len(narrow)} distinct"
          f" for {len(subjects)} subjects")
    print(f"  key (entity, id, gate, criterion, ordinal, basis): {len(wide)} distinct")
    print("  offer B also carries a FRESHNESS reason on its OFFER: the same type"
          " CONFIRM_PRICE, another gate and basis")

    print("\n  T3 the NMI task of Slice 5, over HTTP, on offer A's match:")
    ma = subjects[0][1]
    r = client.post(f"/matches/{ma['match_id']}/review",
                    headers={"Authorization": f"Bearer {ACC_REVIEWER}",
                             "Idempotency-Key": str(uuid.uuid4())},
                    json={"decision": "NEED_MORE_INFORMATION",
                          "reason_code": "PRICE_NEGOTIATION_UNCONFIRMED"})

    print(f"     task_type={task['task_type']} priority={task['priority']} "

          f"criterion={sub.get('criterion')} basis={sub['basis']}")
    r2 = client.post(f"/matches/{ma['match_id']}/review",
                     headers={"Authorization": f"Bearer {ACC_REVIEWER}",

    print("\n  T4 a candidate partial unique index, raced by two REPEATABLE READ"

    key = ("(request_id, (payload->'subject'->>'entity'), (payload->'subject'->>'id'),"
           " (payload->'subject'->>'gate'), coalesce(payload->'subject'->>'criterion',''),"
           " coalesce(payload->'subject'->>'ordinal',''), (payload->'subject'->>'basis'))")

                                     match_id, payload, status)
                  VALUES ('CONFIRM_PRICE', 'HIGH', 't', %(r)s, %(p)s, %(m)s, %(pl)s,

        try:
            a, b = connect(), connect()
            for c in (a, b):
                c.execute("SET TRANSACTION ISOLATION LEVEL REPEATABLE READ")
                c.execute("SELECT 1")
            a.execute(task_ins, args)
            result = {}

            def second():
                try:
                    b.execute(task_ins, args)
                    result["b"] = "ACCEPTED"
                except psycopg.Error as e:
                    result["b"] = f"REFUSED {e.sqlstate} [{e.diag.constraint_name}]"
            th = threading.Thread(target=second)
            th.start()
            time.sleep(0.3)
            waited = one("SELECT %(b)s = ANY(pg_blocking_pids(%(bp)s))",
                         b=a.info.backend_pid, bp=b.info.backend_pid)
            a.commit()
            th.join()
            b.rollback()
            print(f"  WHERE {where}: second insert waited on the first: {waited}; after the"
                  f" first commits: {result['b']}")
            with connect() as c:
                c.execute("UPDATE tasks SET status = 'IN_PROGRESS' WHERE payload = %(pl)s::jsonb",
                          {"pl": payload})
                attempt(c, f"  ... first task IN_PROGRESS, a new OPEN one (WHERE {where[:22]})",
                        task_ins, args)
                c.rollback()
            SETUP.execute("DELETE FROM tasks WHERE payload = %(pl)s::jsonb", {"pl": payload})
            a.close(); b.close()
        finally:
            SETUP.execute("DROP INDEX probe_open_work")

    print("\n  T5 the lock alternative under REPEATABLE READ: B's snapshot is taken"
          " before A commits; B then waits for the lock A held, and checks")
    for lock_name, lock_sql in (
            ("advisory lock on the key", "SELECT pg_advisory_xact_lock(hashtext(%(k)s))"),
            ("request row FOR NO KEY UPDATE",
             "SELECT 1 FROM requests WHERE request_id = %(r)s FOR NO KEY UPDATE")):
        a, b = connect(), connect()
        for c in (a, b):
            c.execute("SET TRANSACTION ISOLATION LEVEL REPEATABLE READ")
            c.execute("SELECT 1")
        lk = {"k": payload, "r": req}
        exists = """SELECT count(*) FROM tasks WHERE payload = %(pl)s::jsonb
                     AND status IN ('OPEN','IN_PROGRESS')"""
        a.execute(lock_sql, lk)
        if not a.execute(exists, {"pl": payload}).fetchone()[0]:
            a.execute(task_ins, args)
        out = {}

        def later():
            try:
                b.execute(lock_sql, lk)
                n = b.execute(exists, {"pl": payload}).fetchone()[0]
                out["seen"] = n
                if not n:
                    b.execute(task_ins, args)
                b.commit()
                out["b"] = "committed"
            except psycopg.Error as e:
                b.rollback()
                out["b"] = f"REFUSED {e.sqlstate}: {e.diag.message_primary}"
        th = threading.Thread(target=later)
        th.start()
        time.sleep(0.3)
        a.commit()
        th.join()
        n = one("SELECT count(*) FROM tasks WHERE payload = %(pl)s::jsonb", pl=payload)
        print(f"  {lock_name:32}: B saw {out.get('seen')} existing task(s); B {out['b']};"
              f" tasks now: {n}")
        SETUP.execute("DELETE FROM tasks WHERE payload = %(pl)s::jsonb", {"pl": payload})
        a.close(); b.close()

    # ===================================================================== E
    section("E  Review point 4 (G6-8): what keeps the evidence of a completion")
    trig = rows("""SELECT tgrelid::regclass::text AS t, tgname FROM pg_trigger
                    WHERE NOT tgisinternal AND tgrelid IN ('turab.observations'::regclass,
                          'turab.evidence_artifacts'::regclass, 'turab.tasks'::regclass,
                          'turab.task_completion_events'::regclass)""")
    print(f"  triggers on observations, evidence_artifacts, tasks, task_completion_events:"
          f" {[(r['t'], r['tgname']) for r in trig] or 'none'}")
    for r in rows("""SELECT conname, conrelid::regclass::text AS t,
                            CASE confdeltype WHEN 'n' THEN 'SET NULL' WHEN 'r' THEN 'RESTRICT'
                                 WHEN 'c' THEN 'CASCADE' WHEN 'a' THEN 'NO ACTION' END AS del
                       FROM pg_constraint
                      WHERE confrelid IN ('turab.observations'::regclass,
                                          'turab.tasks'::regclass)
                      ORDER BY conname"""):
        print(f"  FK {r['conname']:58} on {r['t']:24} ON DELETE {r['del']}")

    def world(c):
        obs = c.execute("""INSERT INTO observations (kind, raw_text, payload,
                                                     recorded_by_account_id)
                           VALUES ('CALL_NOTE', 'seller confirmed land book',
                                   '{"document":"LAND_BOOK"}', %(a)s)
                           RETURNING observation_id""", {"a": ACC_OPERATOR}).fetchone()[0]
        t = c.execute("""INSERT INTO tasks (task_type, priority, title, request_id)
                         VALUES ('VERIFY_DOCUMENT', 'HIGH', 't', %(r)s)
                         RETURNING task_id""", {"r": req}).fetchone()[0]
        e = c.execute("""INSERT INTO task_completion_events (task_id, outcome_code, note,
                             evidence_observation_id, completed_by_account_id)
                         VALUES (%(t)s, 'OTHER', 'called', %(o)s, %(a)s)
                         RETURNING task_completion_event_id""",
                      {"t": t, "o": obs, "a": ACC_OPERATOR}).fetchone()[0]
        c.execute("UPDATE tasks SET status = 'DONE', completed_at = now() WHERE task_id = %s",
                  (t,))
        return obs, t, e

    with connect() as c:
        obs, t, e = world(c)
        print("\n  E1 without an append-only trigger (each statement rolled back):")
        attempt(c, "UPDATE observations SET raw_text, payload (the evidence's body)",
                "UPDATE observations SET raw_text = 'changed', payload = '{}' "
                "WHERE observation_id = %(o)s", {"o": obs})
        c.execute("SAVEPOINT s")
        c.execute("DELETE FROM observations WHERE observation_id = %s", (obs,))
        left = c.execute("SELECT evidence_observation_id FROM task_completion_events "
                         "WHERE task_completion_event_id = %s", (e,)).fetchone()[0]
        c.execute("ROLLBACK TO SAVEPOINT s")
        print(f"  {'DELETE the observation':64} ACCEPTED; the event's"
              f" evidence_observation_id is now {left}")
        attempt(c, "UPDATE task_completion_events SET note",
                "UPDATE task_completion_events SET note = 'x' "
                "WHERE task_completion_event_id = %(e)s", {"e": e})
        attempt(c, "DELETE the completion event",
                "DELETE FROM task_completion_events WHERE task_completion_event_id = %(e)s",
                {"e": e})
        attempt(c, "DELETE the task (it has an event)",
                "DELETE FROM tasks WHERE task_id = %(t)s", {"t": t})
        attempt(c, "UPDATE the DONE task back to OPEN, completed_at NULL",
                "UPDATE tasks SET status = 'OPEN', completed_at = NULL WHERE task_id = %(t)s",
                {"t": t})

        print("\n  E2 with a candidate append-only trigger on task_completion_events"
              " (created in this transaction, rolled back):")
        c.execute("""CREATE FUNCTION pg_temp.probe_append_only() RETURNS trigger
                     LANGUAGE plpgsql AS $$ BEGIN
                       RAISE EXCEPTION 'append-only (probe)'; END $$""")
        c.execute("""CREATE TRIGGER probe_append_only BEFORE UPDATE OR DELETE
                     ON task_completion_events FOR EACH ROW
                     EXECUTE FUNCTION pg_temp.probe_append_only()""")
        attempt(c, "UPDATE task_completion_events SET note",
                "UPDATE task_completion_events SET note = 'x' "
                "WHERE task_completion_event_id = %(e)s", {"e": e})
        attempt(c, "DELETE the observation (its FK action is SET NULL)",
                "DELETE FROM observations WHERE observation_id = %(o)s", {"o": obs})
        attempt(c, "UPDATE observations SET raw_text, payload",
                "UPDATE observations SET raw_text = 'changed', payload = '{}' "
                "WHERE observation_id = %(o)s", {"o": obs})
        c.rollback()

    print("\n  E3 outcome_code against reason_codes:")
    with connect() as c:
        t = c.execute("""INSERT INTO tasks (task_type, title, request_id)
                         VALUES ('OTHER', 't', %(r)s) RETURNING task_id""",
                      {"r": req}).fetchone()[0]
        ev = ("INSERT INTO task_completion_events (task_id, outcome_code,"
              " completed_by_account_id) VALUES (%(t)s, %(c)s, %(a)s)")
        for code in ("NOT_A_CODE", "DEAL_CONFIRMED", "BUYER_REJECTED", None):
            attempt(c, f"outcome_code = {code}", ev, {"t": t, "c": code, "a": ACC_OPERATOR})
        c.execute("UPDATE reason_codes SET active = false WHERE code = 'OTHER'")
        attempt(c, "outcome_code = OTHER, after OTHER is made inactive", ev,
                {"t": t, "c": "OTHER", "a": ACC_OPERATOR})
        c.rollback()
    print("  seeded codes by category:")
    for r in rows("""SELECT category, array_agg(code ORDER BY code) AS codes
                       FROM reason_codes GROUP BY category ORDER BY category"""):
        print(f"    {r['category']:16} {r['codes']}")

    print("\n  E4 two completions of one task, the second waiting on FOR UPDATE:")
    for iso in ("READ COMMITTED", "REPEATABLE READ"):
        t = one("""INSERT INTO tasks (task_type, title, request_id)
                   VALUES ('OTHER', 't', %(r)s) RETURNING task_id""", r=req)
        a, b = connect(), connect()
        for c in (a, b):
            c.execute(f"SET TRANSACTION ISOLATION LEVEL {iso}")
            c.execute("SELECT 1")
        lock = "SELECT status::text FROM tasks WHERE task_id = %s FOR UPDATE"
        a.execute(lock, (t,))
        a.execute("""INSERT INTO task_completion_events (task_id, completed_by_account_id)
                     VALUES (%s, %s)""", (t, ACC_OPERATOR))
        a.execute("UPDATE tasks SET status = 'DONE', completed_at = now() WHERE task_id = %s",
                  (t,))
        out = {}

        def other():
            try:
                out["status"] = b.execute(lock, (t,)).fetchone()[0]
                b.rollback()
            except psycopg.Error as e:
                b.rollback()
                out["status"] = f"REFUSED {e.sqlstate}: {e.diag.message_primary}"
        th = threading.Thread(target=other)
        th.start()
        time.sleep(0.3)
        waited = one("SELECT %(a)s = ANY(pg_blocking_pids(%(b)s))",
                     a=a.info.backend_pid, b=b.info.backend_pid)
        a.commit()
        th.join()
        print(f"  {iso:16}: second waited: {waited}; once the first commits, it reads:"
              f" {out['status']}")
        a.close(); b.close()

    # ===================================================================== S
